# 03-分布式锁与限流实践
> 一句话定位：用一条 `SET NX PX` 写出能上线、能说清边界的分布式锁，并把 Redis 限流接到 Hertz 与 Kitex 上。

- 用 redsync 快速获得多节点锁，并知道它没有解决什么
- 用 Lua 实现令牌桶与滑动窗口，接入 Hertz 中间件与 Kitex 服务端限流
- 用 Redis 预扣库存配合订单落库，避免超卖

## 一、最小可用分布式锁
```go
import "github.com/google/uuid"

type Lock struct {
	rdb   redis.UniversalClient
	key   string
	token string
	ttl   time.Duration
}
func TryLock(ctx context.Context, rdb redis.UniversalClient, key string, ttl time.Duration) (*Lock, bool, error) {
	token := uuid.NewString() // 每次加锁唯一，用于归属校验
	ok, err := rdb.SetNX(ctx, key, token, ttl).Result()
	if err != nil || !ok {
		return nil, false, err
	}
	return &Lock{rdb: rdb, key: key, token: token, ttl: ttl}, true, nil
}
```
三个要素缺一不可：
| 要素 | 作用 | 写错的后果 |
| --- | --- | --- |
| `NX` | 只有 key 不存在才成功 | 后到者覆盖先到者，锁形同虚设 |
| `PX` | 避免进程崩溃后永久死锁 | 一个 panic 锁住整个商品 |
| 唯一 token | 释放时校验归属 | 业务超时后删掉别人的锁 |
## 二、安全释放：必须用 Lua
```go
var releaseScript = redis.NewScript(`
if redis.call("GET", KEYS[1]) == ARGV[1] then
	return redis.call("DEL", KEYS[1])
end
return 0
`)

func (l *Lock) Unlock(ctx context.Context) (bool, error) {
	n, err := releaseScript.Run(ctx, l.rdb, []string{l.key}, l.token).Int64()
	if err != nil {
		return false, err
	}
	return n == 1, nil
}
```
不要写成 `if get == token { del }` 两次命令：GET 与 DEL 之间锁可能已过期并被别人持有，Lua 在 Redis 中单线程执行才能保证比较与删除不可分割。
## 三、续期与看门狗
业务耗时不可预估时（批量导入、跨服务调用），可以让持有者定期把 TTL 续回：
```go
var renewScript = redis.NewScript(`
if redis.call("GET", KEYS[1]) == ARGV[1] then
	return redis.call("PEXPIRE", KEYS[1], ARGV[2])
end
return 0
`)

func (l *Lock) keepAlive(ctx context.Context) {
	ticker := time.NewTicker(l.ttl / 3)
	defer ticker.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case <-ticker.C:
			ok, err := renewScript.Run(context.Background(), l.rdb, []string{l.key}, l.token, l.ttl.Milliseconds()).Int64()
			if err != nil || ok == 0 {
				return // 锁已过期或被抢走，立刻停止续期
			}
		}
	}
}
```
看门狗的边界：
- 续期会掩盖「业务卡死」，锁永远不释放；建议同时设置最长持有时间
- 只有「持锁进程活着但业务慢」才救得了；进程被 kill -9 时看门狗一起消失，锁靠 TTL 兜底
## 四、主从切换丢锁与 Redlock
Redis 主从是**异步复制**：客户端 A 在主节点加锁成功，主节点尚未把 `SET` 同步给从节点就宕机，哨兵把从节点提升为主，客户端 B 立刻能拿到同一把锁。所以单 Redis 锁的准确语义是「尽力而为的互斥」，不是强一致互斥。常见应对：
| 做法 | 说明 | 代价 |
| --- | --- | --- |
| 业务幂等 + 唯一约束 | 数据库唯一键/状态机保证重复执行无副作用 | 最推荐，成本最低 |
| fencing token | 每次加锁返回单调递增序号，下游拒绝旧序号写入 | 需要下游配合 |
| Redlock | 向 N 个独立 Redis 实例过半加锁 | 部署复杂，仍有争议 |
Redlock 的争议来自 Martin Kleppmann 的《How to do distributed locking》与 antirez 的回应：核心矛盾是 GC 停顿/时钟漂移下，客户端可能在锁已失效时仍以为自己持有锁。**务实结论**：普通业务用单实例锁 + 幂等即可；只有「误判代价极高且无法幂等」时才考虑 Redlock，并在架构评审里写明其假设。
## 五、redsync：多节点锁的现成实现
```bash

```
```powershell

```
```go
import (
	goredis "github.com/go-redsync/redsync/v4/redis/goredis/v9"
	"github.com/go-redsync/redsync/v4"
)
pool := goredis.NewPool(rdb) // 单实例也可以；传多个独立实例才是 Redlock
rs := redsync.New(pool)
mutex := rs.NewMutex("mall:lock:order:1001",
	redsync.WithExpiry(8*time.Second),
	redsync.WithTries(3),
	redsync.WithRetryDelay(100*time.Millisecond),
)
if err := mutex.LockContext(ctx); err != nil {
	return fmt.Errorf("acquire lock: %w", err)
}
defer func() { _, _ = mutex.UnlockContext(ctx) }()
// 临界区：创建订单、扣减库存
```
redsync 内部已实现「唯一值 + Lua 释放 + 自动续期」，适合不想自己维护脚本的团队；但它同样无法解决主从切换语义问题，别把它当强一致锁。
## 六、Redis 令牌桶限流
```lua
-- KEYS[1] 限流 key；ARGV: rate(个/秒) capacity now_ms ttl_seconds
local rate = tonumber(ARGV[1])
local capacity = tonumber(ARGV[2])
local now = tonumber(ARGV[3])
local ttl = tonumber(ARGV[4])
local data = redis.call('HMGET', KEYS[1], 'tokens', 'ts')
local tokens = tonumber(data[1])
local ts = tonumber(data[2])
if tokens == nil then
	tokens = capacity
	ts = now
end
local delta = math.max(0, now - ts) / 1000
tokens = math.min(capacity, tokens + delta * rate)
local allowed = 0
if tokens >= 1 then
	tokens = tokens - 1
	allowed = 1
end
redis.call('HMSET', KEYS[1], 'tokens', tokens, 'ts', now)
redis.call('EXPIRE', KEYS[1], ttl)
return allowed
```
```go
var tokenBucket = redis.NewScript(bucketLua)
func Allow(ctx context.Context, rdb redis.UniversalClient, key string, rate, burst float64) (bool, error) {
	n, err := tokenBucket.Run(ctx, rdb, []string{key},
		rate, burst, time.Now().UnixMilli(), 60).Int64()
	return n == 1, err
}
```
令牌桶允许桶容量内的突发，适合网关按用户/租户限流；key 粒度建议 `mall:gw:limit:uid:<uid>`，避免全局一个桶。
## 七、滑动窗口限流
```lua
-- 精确统计最近 window_ms 内的请求数，适合对突发不友好的第三方接口
local now = tonumber(ARGV[1])
local window = tonumber(ARGV[2])
local limit = tonumber(ARGV[3])
redis.call('ZREMRANGEBYSCORE', KEYS[1], 0, now - window)
if redis.call('ZCARD', KEYS[1]) >= limit then
	return 0
end
redis.call('ZADD', KEYS[1], now, now .. '-' .. math.random(1000000))
redis.call('PEXPIRE', KEYS[1], window)
return 1
```
ZSet 方案内存随请求数增长，只适合中小流量；高 QPS 下优先令牌桶或本地限流 + 集群配额。Go 侧同样用 `redis.NewScript` 包装，传入当前毫秒、窗口毫秒数与上限即可。
## 八、接入 Hertz 与 Kitex
```go
// Hertz 网关：按用户维度限流，Redis 故障时放行（fail-open），避免限流组件拖垮入口
func RedisRateLimit(rdb redis.UniversalClient, rate, burst float64) app.HandlerFunc {
	return func(ctx context.Context, c *app.RequestContext) {
		uid := c.GetString("uid")
		if uid == "" {
			uid = c.ClientIP()
		}
		ok, err := Allow(ctx, rdb, "mall:gw:limit:uid:"+uid, rate, burst)
		if err != nil {
			c.Next(ctx) // 限流依赖不可用时放行，并打点
			return
		}
		if !ok {
			c.Header("Retry-After", "1")
			c.AbortWithStatusJSON(429, utils.H{"code": 42900, "msg": "too many requests"})
			return
		}
		c.Next(ctx)
	}
}
```
```go
// Kitex 服务端：单机并发/QPS 兜底，跨实例配额再交给 Redis
svr := product.NewServer(impl, server.WithLimit(&limit.Option{
	MaxConnections: 10000,
	MaxQPS:         3000,
}))
```
分层原则：网关做用户级限流（租户隔离），Kitex 做实例级保护（防单机打满），Redis 做集群级配额；三层阈值从外到内递减，防止流量穿透。
## 九、超卖：Redis 预扣库存 + 订单落库
下单主链路里，库存是并发最激烈的地方。用 Lua 把「判断 + 扣减」做成原子操作，只做**预扣**，最终以 DB 为准：
```go
var deductScript = redis.NewScript(`
local stock = redis.call('GET', KEYS[1])
if not stock then return -1 end
if tonumber(stock) < tonumber(ARGV[1]) then return 0 end
return redis.call('DECRBY', KEYS[1], ARGV[1])
`)
```
```go
// order-service 下单：deductScript 用 redis.NewScript 包装上面的 Lua
left, err := deductScript.Run(ctx, rdb, []string{stockKey(skuID)}, 1).Int64()
if err != nil {
	return nil, err
}
if left <= 0 { // 0 表示库存不足；-1 表示库存未预热
	return nil, ErrSoldOut
}
order := &Order{UserID: uid, SkuID: skuID, Status: StatusPending}
if err := db.WithContext(ctx).Create(order).Error; err != nil {
	_ = rdb.IncrBy(ctx, stockKey(skuID), 1).Err() // 落库失败必须回补
	return nil, err
}
_ = publishStockDeducted(ctx, order) // Kafka 事件，异步更新 MySQL 库存
```
配套要求：
- 订单号/事件 ID 唯一约束，消费者重复扣减 DB 库存时被幂等拦截
- 定时对账任务比对 Redis 预扣、DB 库存、订单明细，修正少卖/超卖
- Redis 预扣 key 与商品库存初始化要做好「预热 + 重建」，Redis 丢失时从 DB 重建而不是从 0 开始
## 常见坑
1. **加锁成功但释放删了别人的锁**：token 用固定值或 `time.Now().String()`，业务一超时就误删。
2. **只 `SETNX` 不设过期**：进程 crash 后锁永久存在，只能人工删 key。
3. **释放分两条命令写**：GET 与 DEL 之间被别人抢占，必须 Lua。
4. **锁 TTL 短于业务耗时**：两个请求同时进临界区，看门狗没开或已退出。
5. **把 Redis 锁当强一致**：主从切换窗口内可能双持有，核心交易必须有 DB 幂等兜底。
## 小结
- 锁 = `SET NX PX` + 唯一 token + Lua 释放，续期只解决业务慢，不解决进程死
- 单实例锁是「尽力互斥」，幂等与唯一约束才是最终防线；Redlock 谨慎使用
- 令牌桶适合网关，滑动窗口适合精确配额的第三方调用
- 限流要分层：网关用户级、Kitex 实例级、Redis 集群级
- 库存用 Lua 预扣换性能，用 DB + 对账换正确性
